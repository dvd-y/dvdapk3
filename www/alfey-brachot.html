הזימון — כשסועדים שלושה ויותר

רַבּוֹתַי נְבָרֵךְ
(המסובים:) יְהִי שֵׁם ה' מְבֹרָךְ מֵעַתָּה וְעַד עוֹלָם
בִּרְשׁוּת מָרָנָן וְרַבּוֹתַי נְבָרֵךְ (אֱלֹהֵינוּ) שֶׁאָכַלְנוּ מִשֶּׁלּוֹ
(המסובים:) בָּרוּךְ (אֱלֹהֵינוּ) שֶׁאָכַלְנוּ מִשֶּׁלּוֹ וּבְטוּבוֹ חָיִינוּ
בעשרה מוסיפים תיבת "אֱלֹהֵינוּ". במאה — "ה' אֱלֹהֵינוּ", ובאלף — "ה' אֱלֹהֵי יִשְׂרָאֵל אֱלֹהֵי הַצְּבָאוֹת".

ברכה א׳ — הַזָּן

בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם
הַזָּן אֶת הָעוֹלָם כֻּלּוֹ בְּטוּבוֹ
בְּחֵן בְּחֶסֶד וּבְרַחֲמִים
הוּא נוֹתֵן לֶחֶם לְכָל בָּשָׂר כִּי לְעוֹלָם חַסְדּוֹ
וּבְטוּבוֹ הַגָּדוֹל תָּמִיד לֹא חָסַר לָנוּ
וְאַל יֶחְסַר לָנוּ מָזוֹן לְעוֹלָם וָעֶד
בַּעֲבוּר שְׁמוֹ הַגָּדוֹל
כִּי הוּא אֵל זָן וּמְפַרְנֵס לַכֹּל
וּמֵטִיב לַכֹּל וּמֵכִין מָזוֹן לְכָל בְּרִיּוֹתָיו אֲשֶׁר בָּרָא
בָּרוּךְ אַתָּה ה' הַזָּן אֶת הַכֹּל

ברכה ב׳ — נוֹדֶה לְּךָ (ברכת הארץ)

נוֹדֶה לְּךָ ה' אֱלֹהֵינוּ עַל שֶׁהִנְחַלְתָּ לַאֲבוֹתֵינוּ
אֶרֶץ חֶמְדָּה טוֹבָה וּרְחָבָה
וְעַל שֶׁהוֹצֵאתָנוּ מֵאֶרֶץ מִצְרַיִם
וּפְדִיתָנוּ מִבֵּית עֲבָדִים
וְעַל בְּרִיתְךָ שֶׁחָתַמְתָּ בִּבְשָׂרֵנוּ
וְעַל תּוֹרָתְךָ שֶׁלִּמַּדְתָּנוּ
וְעַל חֻקֶּיךָ שֶׁהוֹדַעְתָּנוּ
וְעַל חַיִּים חֵן וָחֶסֶד שֶׁחוֹנַנְתָּנוּ
וְעַל אֲכִילַת מָזוֹן שָׁאַתָּה זָן וּמְפַרְנֵס אוֹתָנוּ תָּמִיד
בְּכָל יוֹם וּבְכָל עֵת וּבְכָל שָׁעָה
וְעַל הַכֹּל ה' אֱלֹהֵינוּ אֲנַחְנוּ מוֹדִים לָךְ וּמְבָרְכִים אוֹתָךְ
יִתְבָּרַךְ שִׁמְךָ בְּפִי כָּל חַי תָּמִיד לְעוֹלָם וָעֶד
כַּכָּתוּב: וְאָכַלְתָּ וְשָׂבָעְתָּ וּבֵרַכְתָּ אֶת ה' אֱלֹהֶיךָ
עַל הָאָרֶץ הַטֹּבָה אֲשֶׁר נָתַן לָךְ
בָּרוּךְ אַתָּה ה' עַל הָאָרֶץ וְעַל הַמָּזוֹן

ברכה ג׳ — בּוֹנֵה יְרוּשָׁלַיִם

רַחֵם נָא ה' אֱלֹהֵינוּ עַל יִשְׂרָאֵל עַמֶּךָ
וְעַל יְרוּשָׁלַיִם עִירֶךָ וְעַל צִיּוֹן מִשְׁכַּן כְּבוֹדֶךָ
וְעַל מַלְכוּת בֵּית דָּוִד מְשִׁיחֶךָ
וְעַל הַבַּיִת הַגָּדוֹל וְהַקָּדוֹשׁ שֶׁנִּקְרָא שִׁמְךָ עָלָיו
אֱלֹהֵינוּ אָבִינוּ רְעֵנוּ זוּנֵנוּ פַּרְנְסֵנוּ וְכַלְכְּלֵנוּ
וְהַרְוִיחֵנוּ וְהַרְוַח לָנוּ מְהֵרָה מִכָּל צָרוֹתֵינוּ
וְנָא אַל תַּצְרִיכֵנוּ לֹא לִידֵי מַתְּנַת בָּשָׂר וָדָם
וְלֹא לִידֵי הַלְוָאָתָם
כִּי אִם לְיָדְךָ הַמְּלֵאָה הַפְּתוּחָה הַקְּדוֹשָׁה וְהָרְחָבָה
שֶׁלֹּא נֵבוֹשׁ וְלֹא נִכָּלֵם לְעוֹלָם וָעֶד

(בשבת מוסיפים:)
רְצֵה וְהַחֲלִיצֵנוּ ה' אֱלֹהֵינוּ בְּמִצְוֹתֶיךָ
וּבְמִצְוַת יוֹם הַשְּׁבִיעִי הַשַּׁבָּת הַגָּדוֹל וְהַקָּדוֹשׁ הַזֶּה
כִּי יוֹם זֶה גָּדוֹל וְקָדוֹשׁ הוּא לְפָנֶיךָ
לִשְׁבָּת בּוֹ וְלָנוּחַ בּוֹ בְּאַהֲבָה כְּמִצְוַת רְצוֹנֶךָ

(בראש חודש ובחול המועד מוסיפים יעלה ויבוא:)
אֱלֹהֵינוּ וֵאלֹהֵי אֲבוֹתֵינוּ, יַעֲלֶה וְיָבֹא, וְיַגִּיעַ, וְיֵרָאֶה וְיֵרָצֶה וְיִשָּׁמַע, וְיִפָּקֵד וְיִזָּכֵר זִכְרוֹנֵנוּ וּפִקְדוֹנֵנוּ... בְּיוֹם רֹאשׁ הַחֹדֶשׁ הַזֶּה. זָכְרֵנוּ ה' אֱלֹהֵינוּ בּוֹ לְטוֹבָה, וּפָקְדֵנוּ בוֹ לִבְרָכָה, וְהוֹשִׁיעֵנוּ בוֹ לְחַיִּים

וּבְנֵה יְרוּשָׁלַיִם עִיר הַקֹּדֶשׁ בִּמְהֵרָה בְיָמֵינוּ
בָּרוּךְ אַתָּה ה' בּוֹנֵה בְרַחֲמָיו יְרוּשָׁלָיִם אָמֵן

ברכה ד׳ — הַטּוֹב וְהַמֵּטִיב

בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם
הָאֵל אָבִינוּ מַלְכֵּנוּ אַדִּירֵנוּ בּוֹרְאֵנוּ גֹּאֲלֵנוּ
יוֹצְרֵנוּ קְדוֹשֵׁנוּ קְדוֹשׁ יַעֲקֹב
רוֹעֵנוּ רוֹעֵה יִשְׂרָאֵל
הַמֶּלֶךְ הַטּוֹב וְהַמֵּטִיב לַכֹּל
שֶׁבְּכָל יוֹם וָיוֹם הוּא הֵטִיב הוּא מֵטִיב הוּא יֵיטִיב לָנוּ
הוּא גְמָלָנוּ הוּא גוֹמְלֵנוּ הוּא יִגְמְלֵנוּ לָעַד
לְחֵן וּלְחֶסֶד וּלְרַחֲמִים וּלְרֶוַח
הַצָּלָה וְהַצְלָחָה בְּרָכָה וִישׁוּעָה נֶחָמָה
פַּרְנָסָה וְכַלְכָּלָה וְרַחֲמִים וְחַיִּים וְשָׁלוֹם וְכָל טוֹב
וּמִכָּל טוּב לְעוֹלָם אַל יְחַסְּרֵנוּ

הָרַחֲמָן — בקשות

הָרַחֲמָן הוּא יִמְלֹךְ עָלֵינוּ לְעוֹלָם וָעֶד
הָרַחֲמָן הוּא יִתְבָּרַךְ בַּשָּׁמַיִם וּבָאָרֶץ
הָרַחֲמָן הוּא יִשְׁתַּבַּח לְדוֹר דּוֹרִים
וְיִתְפָּאַר בָּנוּ לָעַד וּלְנֵצַח נְצָחִים
הָרַחֲמָן הוּא יְפַרְנְסֵנוּ בְּכָבוֹד
הָרַחֲמָן הוּא יִשְׁבֹּר עֻלֵּנוּ מֵעַל צַוָּארֵנוּ
וְהוּא יוֹלִיכֵנוּ קוֹמְמִיּוּת לְאַרְצֵנוּ
הָרַחֲמָן הוּא יִשְׁלַח בְּרָכָה מְרֻבָּה בַּבַּיִת הַזֶּה
וְעַל שֻׁלְחָן זֶה שֶׁאָכַלְנוּ עָלָיו
הָרַחֲמָן הוּא יִשְׁלַח לָנוּ אֶת אֵלִיָּהוּ הַנָּבִיא זָכוּר לַטּוֹב
וִיבַשֶּׂר לָנוּ בְּשׂוֹרוֹת טוֹבוֹת יְשׁוּעוֹת וְנֶחָמוֹת
הָרַחֲמָן הוּא יְבָרֵךְ אֶת כָּל הַמְּסֻבִּים כָּאן
אוֹתָנוּ וְאֶת כָּל אֲשֶׁר לָנוּ
כְּמוֹ שֶׁנִּתְבָּרְכוּ אֲבוֹתֵינוּ אַבְרָהָם יִצְחָק וְיַעֲקֹב
בַּכֹּל מִכֹּל כֹּל כֵּן יְבָרֵךְ אוֹתָנוּ כֻּלָּנוּ יַחַד
בִּבְרָכָה שְׁלֵמָה וְנֹאמַר אָמֵן
בבית ההורים מוסיפים "אֶת אָבִי מוֹרִי... וְאֶת אִמִּי מוֹרָתִי", ובביתו מוסיף "אוֹתִי וְאֶת אִשְׁתִּי וְאֶת זַרְעִי וְאֶת כָּל אֲשֶׁר לִי".

מעין שלוש — נוסח מזרחי (על המחיה)

בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם
עַל הַמִּחְיָה וְעַל הַכַּלְכָּלָה
וְעַל תְּנוּבַת הַשָּׂדֶה
וְעַל אֶרֶץ חֶמְדָּה טוֹבָה וּרְחָבָה
שֶׁרָצִיתָ וְהִנְחַלְתָּ לַאֲבוֹתֵינוּ לֶאֱכֹל מִפִּרְיָהּ וְלִשְׂבֹּעַ מִטּוּבָהּ
רַחֵם נָא ה' אֱלֹהֵינוּ עַל יִשְׂרָאֵל עַמֶּךָ
וְעַל יְרוּשָׁלַיִם עִירֶךָ
וְעַל צִיּוֹן מִשְׁכַּן כְּבוֹדֶךָ
וְעַל מִזְבַּחֶךָ וְעַל הֵיכָלֶךָ
וּבְנֵה יְרוּשָׁלַיִם עִיר הַקֹּדֶשׁ בִּמְהֵרָה בְיָמֵינוּ
וְהַעֲלֵנוּ לְתוֹכָהּ וְשַׂמְּחֵנוּ בְּבִנְיָנָהּ
וְנֹאכַל מִפִּרְיָהּ וְנִשְׂבַּע מִטּוּבָהּ וּנְבָרֶכְךָ עָלֶיהָ בִּקְדֻשָּׁה וּבְטָהֳרָה
(בשבת:) וּרְצֵה וְהַחֲלִיצֵנוּ בְּיוֹם הַשַּׁבָּת הַזֶּה
כִּי אַתָּה ה' טוֹב וּמֵטִיב לַכֹּל וְנוֹדֶה לְּךָ עַל הָאָרֶץ
בָּרוּךְ אַתָּה ה' עַל הָאָרֶץ וְעַל הַמִּחְיָה
באותה מטבע משתנה הפתיחה והחתימה לפי סוג המאכל: "עַל הָעֵץ וְעַל פְּרִי הָעֵץ...וְעַל הַפֵּרוֹת" לפירות משבעת המינים, ו"עַל הַגֶּפֶן וְעַל פְּרִי הַגֶּפֶן...וְעַל פְּרִי הַגָּפֶן" ליין ומיץ ענבים.

בורא נפשות רבות

בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם
בּוֹרֵא נְפָשׁוֹת רַבּוֹת וְחֶסְרוֹנָן
עַל כָּל מַה שֶּׁבָּרָאתָ לְהַחֲיוֹת בָּהֶם נֶפֶשׁ כָּל חָי
בָּרוּךְ חֵי הָעוֹלָמִים
מברכים לאחר אכילת פירות וירקות (שאינם משבעת המינים), בשר, דגים, ביצים, מוצרי חלב, ממתקים ורוב המשקאות, בשיעור המחייב ברכה אחרונה.

ברכות השולחן — ברכות ראשונות

המוציא לחם מן הארץ:
בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם הַמּוֹצִיא לֶחֶם מִן הָאָרֶץ
בורא פרי הגפן:
בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם בּוֹרֵא פְּרִי הַגָּפֶן
בורא מיני מזונות:
בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם בּוֹרֵא מִינֵי מְזוֹנוֹת
על פירות העץ:
בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם בּוֹרֵא פְּרִי הָעֵץ
על פירות האדמה:
בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם בּוֹרֵא פְּרִי הָאֲדָמָה

ברכה ג׳ — בּוֹנֵה יְרוּשָׁלַיִם

וְעַל יְרוּשָׁלַיִם עִירֶךָ וְעַל צִיּוֹן מִשְׁכַּן כְּבוֹדֶךָ
וְעַל מַלְכוּת בֵּית דָּוִד מְשִׁיחֶךָ

יוֹצְרֵנוּ קְדוֹשֵׁנוּ קְדוֹשׁ יַעֲקֹב רוֹעֵנוּ רוֹעֵה יִשְׂרָאֵל

נטילת ידיים לסעודה

בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם
אֲשֶׁר קִדְּשָׁנוּ בְּמִצְוֹתָיו וְצִוָּנוּ עַל נְטִילַת יָדָיִם

אשר יצר — לאחר יציאה מהשירותים

בָּרוּךְ אַתָּה ה' אֱלֹהֵינוּ מֶלֶךְ הָעוֹלָם
אֲשֶׁר יָצַר אֶת הָאָדָם בְּחָכְמָה
וּבָרָא בוֹ נְקָבִים נְקָבִים חֲלוּלִים חֲלוּלִים
גָּלוּי וְיָדוּעַ לִפְנֵי כִסֵּא כְבוֹדֶךָ
שֶׁאִם יִסָּתֵם אֶחָד מֵהֶם אוֹ אִם יִפָּתֵחַ אֶחָד מֵהֶם
אִי אֶפְשָׁר לְהִתְקַיֵּם וְלַעֲמוֹד לְפָנֶיךָ אֲפִילוּ שָׁעָה אֶחָת
בָּרוּךְ אַתָּה ה' רוֹפֵא כָל בָּשָׂר וּמַפְלִיא לַעֲשׂוֹת
